# 使用Qt5开发跨平台串口调试助手

> 作者：蔡浩宇 | Qt开发笔记

## 引言

串口调试助手是嵌入式开发者日常必备工具，但市面上的软件往往功能臃肿或不符合个人习惯。使用Qt开发一款专属串口工具，既能满足定制化需求，又是学习Qt串口编程、多线程、QSS样式表的综合实战项目。本文将完整实现一个功能实用的串口调试助手。

## 2. 串口管理类

### 2.1 SerialManager.h

```cpp
#ifndef SERIALMANAGER_H
#define SERIALMANAGER_H

#include <QObject>
#include <QSerialPort>
#include <QSerialPortInfo>

class SerialManager : public QObject
{
    Q_OBJECT
public:
    explicit SerialManager(QObject *parent = nullptr);

    /* 扫描可用串口 */
    QStringList availablePorts();

    /* 打开/关闭串口 */
    bool open(const QString &portName,
              qint32 baudRate = 115200,
              QSerialPort::DataBits dataBits = QSerialPort::Data8,
              QSerialPort::Parity parity = QSerialPort::NoParity,
              QSerialPort::StopBits stopBits = QSerialPort::OneStop,
              QSerialPort::FlowControl flow = QSerialPort::NoFlowControl);
    void close();

    /* 发送数据 */
    qint64 sendText(const QString &text);
    qint64 sendHex(const QString &hexString);

    bool isOpen() const;

    /* 统计 */
    quint64 rxBytes() const { return m_rxBytes; }
    quint64 txBytes() const { return m_txBytes; }

signals:
    void dataReceived(const QByteArray &data);
    void portOpened(const QString &portName);
    void portClosed();
    void errorOccurred(const QString &error);

private slots:
    void handleReadyRead();
    void handleError(QSerialPort::SerialPortError error);

private:
    QSerialPort *m_serial;
    quint64 m_rxBytes = 0;
    quint64 m_txBytes = 0;
};

#endif // SERIALMANAGER_H
```

### 2.2 SerialManager.cpp

```cpp
#include "serialmanager.h"

SerialManager::SerialManager(QObject *parent)
    : QObject(parent), m_serial(new QSerialPort(this))
{
    connect(m_serial, &QSerialPort::readyRead,
            this, &SerialManager::handleReadyRead);
    connect(m_serial, &QSerialPort::errorOccurred,
            this, &SerialManager::handleError);
}

QStringList SerialManager::availablePorts()
{
    QStringList ports;
    for (const QSerialPortInfo &info : QSerialPortInfo::availablePorts())
    {
        ports << info.portName() + " (" + info.description() + ")";
    }
    return ports;
}

bool SerialManager::open(const QString &portName, qint32 baudRate,
                          QSerialPort::DataBits dataBits,
                          QSerialPort::Parity parity,
                          QSerialPort::StopBits stopBits,
                          QSerialPort::FlowControl flow)
{
    if (m_serial->isOpen())
        m_serial->close();

    m_serial->setPortName(portName);
    m_serial->setBaudRate(baudRate);
    m_serial->setDataBits(dataBits);
    m_serial->setParity(parity);
    m_serial->setStopBits(stopBits);
    m_serial->setFlowControl(flow);

    if (m_serial->open(QIODevice::ReadWrite))
    {
        emit portOpened(portName);
        return true;
    }

    emit errorOccurred(m_serial->errorString());
    return false;
}

void SerialManager::close()
{
    if (m_serial->isOpen())
    {
        m_serial->close();
        emit portClosed();
    }
}

void SerialManager::handleReadyRead()
{
    QByteArray data = m_serial->readAll();
    m_rxBytes += data.size();
    emit dataReceived(data);
}

void SerialManager::handleError(QSerialPort::SerialPortError error)
{
    if (error != QSerialPort::NoError && error != QSerialPort::TimeoutError)
    {
        emit errorOccurred(m_serial->errorString());
    }
}

qint64 SerialManager::sendText(const QString &text)
{
    QByteArray data = text.toUtf8();
    qint64 written = m_serial->write(data);
    if (written > 0)
    {
        m_serial->flush();
        m_txBytes += written;
    }
    return written;
}

qint64 SerialManager::sendHex(const QString &hexString)
{
    QByteArray data = QByteArray::fromHex(hexString.toUtf8());
    qint64 written = m_serial->write(data);
    if (written > 0)
    {
        m_serial->flush();
        m_txBytes += written;
    }
    return written;
}

bool SerialManager::isOpen() const
{
    return m_serial->isOpen();
}
```

## 3. 主界面实现

### 3.1 界面布局

```cpp
// MainWindow::MainWindow 构造函数中
MainWindow::MainWindow(QWidget *parent) : QMainWindow(parent)
{
    setWindowTitle("Serial Assistant");
    setMinimumSize(800, 600);

    /* ---- 串口配置区 ---- */
    QGroupBox *configGroup = new QGroupBox("Port Config");

    m_portCombo = new QComboBox();
    m_baudCombo = new QComboBox();
    m_baudCombo->addItems({"9600", "19200", "38400", "57600", "115200", "921600"});
    m_baudCombo->setCurrentText("115200");

    m_openBtn = new QPushButton("Open");
    m_openBtn->setCheckable(true);

    QHBoxLayout *configLayout = new QHBoxLayout(configGroup);
    configLayout->addWidget(new QLabel("Port:"));
    configLayout->addWidget(m_portCombo);
    configLayout->addWidget(new QLabel("Baud:"));
    configLayout->addWidget(m_baudCombo);
    configLayout->addWidget(m_openBtn);

    /* ---- 接收显示区 ---- */
    QGroupBox *rxGroup = new QGroupBox("Received Data");
    m_rxText = new QTextEdit();
    m_rxText->setReadOnly(true);
    QVBoxLayout *rxLayout = new QVBoxLayout(rxGroup);
    rxLayout->addWidget(m_rxText);

    /* ---- 发送区 ---- */
    QGroupBox *txGroup = new QGroupBox("Send Data");
    m_txInput = new QLineEdit();
    m_txInput->setPlaceholderText("Enter data to send...");
    m_sendBtn = new QPushButton("Send");
    m_sendBtn->setEnabled(false);
    QHBoxLayout *txLayout = new QHBoxLayout(txGroup);
    txLayout->addWidget(m_txInput);
    txLayout->addWidget(m_sendBtn);

    /* ---- 主布局 ---- */
    QWidget *central = new QWidget();
    QVBoxLayout *mainLayout = new QVBoxLayout(central);
    mainLayout->addWidget(configGroup);
    mainLayout->addWidget(rxGroup, 1);     // 接收区占更多空间
    mainLayout->addWidget(txGroup);

    setCentralWidget(central);

    /* ---- 状态栏 ---- */
    m_statusLabel = new QLabel("Disconnected");
    statusBar()->addPermanentWidget(m_statusLabel);
    statusBar()->addPermanentWidget(new QLabel("  |  "));
    m_countLabel = new QLabel("RX: 0  TX: 0");
    statusBar()->addPermanentWidget(m_countLabel);
}
```

### 3.2 信号连接

```cpp
/* 在MainWindow构造函数中连接信号槽 */
connect(m_serial, &SerialManager::dataReceived,
        this, &MainWindow::onDataReceived);

connect(m_serial, &SerialManager::portOpened, [&](const QString &name) {
    m_openBtn->setText("Close");
    m_sendBtn->setEnabled(true);
    m_statusLabel->setText("Connected: " + name);
    m_statusLabel->setStyleSheet("color: green; font-weight: bold;");
});

connect(m_serial, &SerialManager::portClosed, [&]() {
    m_openBtn->setText("Open");
    m_sendBtn->setEnabled(false);
    m_statusLabel->setText("Disconnected");
    m_statusLabel->setStyleSheet("color: red;");
});

connect(m_openBtn, &QPushButton::clicked, this, &MainWindow::togglePort);
connect(m_sendBtn, &QPushButton::clicked, this, &MainWindow::sendData);
connect(m_txInput, &QLineEdit::returnPressed, this, &MainWindow::sendData);
```

### 3.3 数据处理

```cpp
void MainWindow::onDataReceived(const QByteArray &data)
{
    m_rxText->moveCursor(QTextCursor::End);

    if (m_hexDisplay)
    {
        /* HEX显示模式 */
        QString hex = data.toHex(' ').toUpper();
        m_rxText->insertPlainText(hex + "\n");
    }
    else
    {
        /* 文本显示模式 */
        m_rxText->insertPlainText(QString::fromUtf8(data));
    }

    m_rxText->moveCursor(QTextCursor::End);
    m_countLabel->setText(
        QString("RX: %1  TX: %2")
            .arg(m_serial->rxBytes())
            .arg(m_serial->txBytes())
    );
}

void MainWindow::sendData()
{
    if (!m_serial->isOpen()) return;

    QString text = m_txInput->text();
    if (!text.isEmpty())
    {
        if (m_hexSend)
        {
            m_serial->sendHex(text);
        }
        else
        {
            m_serial->sendText(text);
        }
        m_txInput->clear();
    }
}
```
