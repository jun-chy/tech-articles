# 哈希表原理与C语言实现：从冲突处理到工业实践

> 作者：蔡浩宇 | 数据结构与算法笔记

## 引言

哈希表（Hash Table）是计算机科学中最伟大的数据结构之一。无论是编译器的符号表、数据库的索引结构，还是Python/JavaScript中的字典对象，哈希表都是底层核心实现。在嵌入式系统中，哈希表常用于命令解析、设备注册表查找等O(1)场景。本文将从零实现一个工业级哈希表，深入讨论哈希函数选择和冲突处理策略。

---

## 1. 哈希表核心概念

### 1.1 工作原理

```
键(Key) ──► 哈希函数 ──► 哈希值(索引) ──► 存储槽(Bucket)

示例：
"temperature" ──► hash("temperature") ──► 7 ──► bucket[7] ──► {key, value}
"humidity"    ──► hash("humidity")    ──► 3 ──► bucket[3] ──► {key, value}
"pressure"    ──► hash("pressure")    ──► 7 ──► bucket[7] ──► 冲突！

理想情况：每个键映射到唯一索引 → O(1) 查找
实际情况：不同键可能映射到相同索引 → 需要冲突处理
```

### 1.2 时间复杂度

| 操作 | 平均 | 最坏 |
|------|------|------|
| 插入 | O(1) | O(n) |
| 查找 | O(1) | O(n) |
| 删除 | O(1) | O(n) |

## 2. 哈希函数设计

### 2.1 常见哈希函数对比

```c
/* 方法1：简单取余法（适用于整数键） */
uint32_t hash_mod(uint32_t key, uint32_t size)
{
    return key % size;
}

/* 方法2：乘法哈希法（Knuth推荐） */
uint32_t hash_multiply(uint32_t key, uint32_t size)
{
    // A = 0.6180339887... (黄金比例的小数部分)
    return (uint32_t)(size * ((double)key * 0.6180339887) % 1.0);
}

/* 方法3：DJB2字符串哈希（经典且高效） */
uint32_t hash_djb2(const char *str, uint32_t size)
{
    uint32_t hash = 5381;
    while (*str)
    {
        hash = ((hash << 5) + hash) + (uint8_t)*str;  // hash * 33 + c
        str++;
    }
    return hash % size;
}

/* 方法4：FNV-1a（分布更均匀） */
uint32_t hash_fnv1a(const char *str, uint32_t size)
{
    uint32_t hash = 2166136261u;  // FNV offset basis
    while (*str)
    {
        hash ^= (uint8_t)*str;
        hash *= 16777619u;        // FNV prime
        str++;
    }
    return hash % size;
}
```

> **选型建议**：嵌入式场景下，DJB2的 `hash * 33` 等价于 `(hash << 5) + hash`，无乘法运算，适合无FPU的MCU。

## 3. 冲突处理策略

### 3.1 开放寻址法（线性探测）

```c
/*
  冲突时线性向后探测下一个空槽：

  插入 "pressure" 时，hash("pressure") = 7，但 bucket[7] 已被占用
  → 检查 bucket[8] → 如果也被占用
  → 检查 bucket[9] → ...
  → 直到找到空槽或表已满
*/

#define HASH_TABLE_SIZE  16

typedef struct {
    char    key[32];
    int     value;
    uint8_t occupied;  // 0=空, 1=已占用, 2=已删除(墓碑)
} HashEntry;

typedef struct {
    HashEntry entries[HASH_TABLE_SIZE];
    uint16_t count;     // 当前元素数量
} HashTable_Linear;

uint32_t hash_string(const char *str)
{
    uint32_t hash = 5381;
    while (*str)
    {
        hash = ((hash << 5) + hash) + (uint8_t)*str;
        str++;
    }
    return hash % HASH_TABLE_SIZE;
}

/* 线性探测插入 */
int Hash_Insert(HashTable_Linear *ht, const char *key, int value)
{
    if (ht->count >= HASH_TABLE_SIZE - 1)
        return -1;  // 表已满

    uint32_t idx = hash_string(key);

    for (uint32_t i = 0; i < HASH_TABLE_SIZE; i++)
    {
        uint32_t pos = (idx + i) % HASH_TABLE_SIZE;

        if (ht->entries[pos].occupied != 1)  // 空槽或墓碑
        {
            strncpy(ht->entries[pos].key, key, 31);
            ht->entries[pos].value = value;
            ht->entries[pos].occupied = 1;
            ht->count++;
            return 0;
        }

        /* 已存在相同key → 更新值 */
        if (strcmp(ht->entries[pos].key, key) == 0)
        {
            ht->entries[pos].value = value;
            return 0;
        }
    }
    return -1;  // 遍历完仍未找到空槽
}

/* 线性探测查找 */
int Hash_Find(HashTable_Linear *ht, const char *key, int *value)
{
    uint32_t idx = hash_string(key);

    for (uint32_t i = 0; i < HASH_TABLE_SIZE; i++)
    {
        uint32_t pos = (idx + i) % HASH_TABLE_SIZE;

        if (ht->entries[pos].occupied == 0)
            return -1;  // 遇到空槽，说明key不存在

        if (ht->entries[pos].occupied == 1 &&
            strcmp(ht->entries[pos].key, key) == 0)
        {
            *value = ht->entries[pos].value;
            return 0;
        }
    }
    return -1;
}
```

### 3.2 链地址法（拉链法）

```c
/* 每个桶维护一个链表 */

typedef struct ChainNode {
    char key[32];
    int  value;
    struct ChainNode *next;
} ChainNode;

typedef struct {
    ChainNode *buckets[HASH_TABLE_SIZE];
    uint16_t   count;
} HashTable_Chain;

/* 链地址法插入 */
int HashChain_Insert(HashTable_Chain *ht, const char *key, int value)
{
    uint32_t idx = hash_string(key);

    /* 先查找是否已存在 */
    ChainNode *cur = ht->buckets[idx];
    while (cur != NULL)
    {
        if (strcmp(cur->key, key) == 0)
        {
            cur->value = value;  // 更新
            return 0;
        }
        cur = cur->next;
    }

    /* 创建新节点插入链表头部 */
    ChainNode *node = (ChainNode *)malloc(sizeof(ChainNode));
    if (!node) return -1;
    strncpy(node->key, key, 31);
    node->value = value;
    node->next = ht->buckets[idx];
    ht->buckets[idx] = node;
    ht->count++;
    return 0;
}
```

### 3.3 两种策略对比

```
开放寻址法（线性探测）：
┌───┬───┬───┬───┬───┬───┬───┬───┐
│ A │ B │ D │ F │   │ C │   │ E │
└───┴───┴───┴───┴───┴───┴───┴───┘
  0   1   2   3   4   5   6   7

  优点：缓存友好（连续内存），无额外内存开销
  缺点：聚集效应，删除需要墓碑标记

链地址法（拉链法）：
┌───┐   ┌───┬───┐   ┌───┐
│ A │──►│ B │→│ G │   │ F │
└───┘   └───┴───┘   └───┘
  0       1           3

  优点：不怕冲突，删除简单
  缺点：指针开销，缓存不友好
```

## 4. 嵌入式实战：命令解析表

```c
/* 使用哈希表实现高效的命令解析 */
typedef void (*CmdHandler)(int argc, char *argv[]);

typedef struct {
    char       name[16];
    CmdHandler handler;
} CmdEntry;

/* 预定义的命令表（静态哈希，编译期确定） */
static const CmdEntry cmd_table[] = {
    {"help",    Cmd_Help},
    {"reset",   Cmd_Reset},
    {"status",  Cmd_Status},
    {"config",  Cmd_Config},
    {"read",    Cmd_Read},
    {"write",   Cmd_Write},
};
#define CMD_COUNT  (sizeof(cmd_table) / sizeof(cmd_table[0]))

/* 查找命令 - O(1)平均 */
CmdHandler FindCommand(const char *name)
{
    uint32_t idx = hash_djb2(name, CMD_COUNT);
    if (strcmp(cmd_table[idx].name, name) == 0)
    {
        return cmd_table[idx].handler;
    }
    /* 简单线性探测后备 */
    for (uint32_t i = 1; i < CMD_COUNT; i++)
    {
        uint32_t pos = (idx + i) % CMD_COUNT;
        if (cmd_table[pos].name[0] == '\0') break;
        if (strcmp(cmd_table[pos].name, name) == 0)
            return cmd_table[pos].handler;
    }
    return NULL;  // 未找到
}
```

## 5. 装载因子与性能

```
性能 vs 装载因子(α = n/m):

查找次数          开放寻址    链地址法
α = 0.5          1.5        1.0
α = 0.75         2.5        1.25
α = 0.9          5.5        1.45
α = 1.0         → ∞        1.0 (退化为链表遍历)

建议：开放寻址法 α < 0.7，链地址法 α < 0.9
嵌入式场景：使用固定大小的表，避免动态扩容的开销
```

## 6. 总结

| 方面 | 建议 |
|------|------|
| 哈希函数 | 整数用取余，字符串用DJB2或FNV-1a |
| 冲突处理 | 嵌入式优先开放寻址法（无malloc） |
| 表大小 | 取素数或2^n，根据最大元素数预估 |
| 装载因子 | 控制在0.7以下以保证性能 |
| 内存管理 | 嵌入式场景优先静态分配 |

哈希表是"用空间换时间"的经典案例，理解其实现细节有助于在实际项目中做出正确的工程决策。
